import numpy as np 
import matplotlib.pyplot as plt 
from tensorflow.keras.datasets import imdb 
from tensorflow.keras.models import Sequential 
from tensorflow.keras.layers import Dense 
# 1. Load Data & Vectorize 
NUM_WORDS = 10000 
(x_train_raw, y_train), (x_test_raw, y_test) = imdb.load_data(num_words=NUM_WORDS) 
def vectorize(seqs, dim=NUM_WORDS): 
    results = np.zeros((len(seqs), dim)) 
    for i, s in enumerate(seqs): 
        results[i, s] = 1.0 
    return results 
x_train, x_test = vectorize(x_train_raw), vectorize(x_test_raw) 
y_train, y_test = np.asarray(y_train).astype('float32'), np.asarray(y_test).astype('float32') 
# 2. Build & Compile Model 
model = Sequential([ 
    Dense(16, activation='relu', input_shape=(NUM_WORDS,)), 
    Dense(16, activation='relu'), 
    Dense(1, activation='sigmoid') 
]) 
model.compile(optimizer='adam', loss='binary_crossentropy', metrics=['accuracy']) 
# 3. Train Model 
history = model.fit(x_train, y_train, epochs=8, batch_size=512, validation_split=0.2, 
verbose=1)

import matplotlib.pyplot as plt 
from tensorflow.keras.datasets import boston_housing 
from tensorflow.keras.models import Sequential 
from tensorflow.keras.layers import Dense 
 
# Data load & normalize 
(x_tr, y_tr), (x_te, y_te) = boston_housing.load_data() 
x_tr = (x_tr - x_tr.mean(axis=0)) / x_tr.std(axis=0) 
x_te = (x_te - x_te.mean(axis=0)) / x_te.std(axis=0) 
 
# Build & train model 
model = Sequential([Dense(64, 'relu', input_shape=(13,)), Dense(64, 'relu'), Dense(1)]) 
model.compile('adam', loss='mse') 
history = model.fit(x_tr, y_tr, validation_data=(x_te, y_te), epochs=80, batch_size=16, 
verbose=0) 
 
# Plot Training & Testing Loss Graph 
plt.plot(history.history['loss'], label='Train Loss') 
plt.plot(history.history['val_loss'], label='Validation Loss') 
plt.title('Boston Housing Price Loss Graph') 
plt.xlabel('Epochs') 
plt.ylabel('Loss (MSE)') 
plt.legend() 
plt.show()

Exp-6:- 
Build a Convolution Neural Network for simple  image(dogs and cats)classification. 
import tensorflow as tf 
from tensorflow.keras import layers, Sequential 
import matplotlib.pyplot as plt 
import numpy as np 
# CIFAR10 data load & Cat (3), Dog (5) filter 
(x_train, y_train), (x_test, y_test) = tf.keras.datasets.cifar10.load_data() 
idx = np.where((y_train == 3) | (y_train == 5))[0][:300] 
idx_t = np.where((y_test == 3) | (y_test == 5))[0][:10] 
x_tr, y_tr = x_train[idx], (y_train[idx] == 5).astype(int) 
x_te, y_te = x_test[idx_t], (y_test[idx_t] == 5).astype(int) 
model = Sequential([ 
    layers.Rescaling(1./255, input_shape=(32, 32, 3)), 
    layers.Conv2D(16, 3, activation="relu"), 
    layers.MaxPooling2D(), 
    layers.Flatten(), 
    layers.Dense(1, activation="sigmoid") 
]) 
model.compile(optimizer="adam", loss="binary_crossentropy", metrics=["accuracy"]) 
model.fit(x_tr, y_tr, epochs=3, batch_size=32, verbose=0) 
pred = (model.predict(x_te) > 0.5).astype(int) 
plt.plot(y_te, 'ro', label='Actual') 
plt.plot(pred, 'bx', label='Predicted') 
plt.yticks([0, 1], ['Cat', 'Dog']) 
plt.title('Dog vs Cat Classification') 
plt.legend() 
plt.show()

Use a pre-trained convolution neural network(VGG16) for image 
classification. 
import numpy as np 
import matplotlib.pyplot as plt 
from tensorflow.keras.applications.vgg16 import VGG16, preprocess_input, 
decode_predictions 
# 1. Model & Random Image setup 
model = VGG16(weights='imagenet') 
img = np.random.randint(0, 255, (1, 224, 224, 3), dtype=np.uint8) 
# 2. Prediction 
preds = model.predict(preprocess_input(img)) 
decoded = decode_predictions(preds, top=3)[0] 
# Print Predictions 
for i, (_, label, score) in enumerate(decoded): 
    print(f"{i + 1}. {label}: {score * 100:.2f}%") 
# 3. Bar Chart Visualization (Mee Graph Alaane Undhi) 
labels = [item[1] for item in decoded] 
scores = [item[2] * 100 for item in decoded] 
plt.figure(figsize=(7, 4)) 
plt.barh(labels, scores, color=['#3498db', '#2ecc71', '#e74c3c']) 
plt.xlabel('Probability (%)') 
plt.title('VGG16 Classification Prediction Probabilities', fontweight='bold') 
plt.gca().invert_yaxis() 
plt.grid(axis='x', linestyle='--', alpha=0.7)

Exp-8:- 
Implement one hot encoding of words or characters. 
import numpy as np 
import matplotlib.pyplot as plt 
import seaborn as sns 
# 1. Input Sentences (Text Data) 
sentences = ["deep learning lab", "neural network model"] 
# 2. Extract Unique Words (Vocabulary) 
words = sorted(list(set(" ".join(sentences).split()))) 
vocab_size = len(words) 
word_to_index = {word: i for i, word in enumerate(words)} 
# 3. Create One-Hot Encodings Matrix 
one_hot_matrix = np.zeros((len(words), vocab_size)) 
for i, word in enumerate(words): 
    one_hot_matrix[i, word_to_index[word]] = 1 
# Display Word to One-Hot Vector Mapping 
print("Vocabulary Mapping:") 
for word, idx in word_to_index.items(): 
    print(f"'{word}': {one_hot_matrix[idx].astype(int)}") 
# 4. Visualization: One-Hot Encoding Heatmap Matrix 
plt.figure(figsize=(7, 5)) 
sns.heatmap(one_hot_matrix, annot=True, cbar=False, cmap="Blues", 
            xticklabels=words, yticklabels=words, fmt=".0f", linewidths=1) 
 
plt.title("One-Hot Encoding Matrix Representation", fontweight="bold") 
plt.xlabel("Encoded Vector Index (One-Hot)") 
plt.ylabel("Words") 
plt.show()

(x_train, y_train), (x_test, y_test) = tf.keras.datasets.imdb.load_data(

)

x_train = tf.keras.utils.pad_sequences(x_train, maxlen=100) 
x_test = tf.keras.utils.pad_sequences(x_test, maxlen=100) 
# Word Embedding Model

    tf.keras.layers.GlobalAveragePooling1D(), 
    tf.keras.layers.Dense(1, activation="sigmoid") 
]) 
model.compile( 
    optimizer="adam", 
    loss="binary_crossentropy", 
    metrics=["accuracy"] 
)
